High-Throughput Storage Architecture

Microsoft DirectStorage Architecture

Modern game asset streaming has outgrown the traditional synchronous storage stack. DirectStorage replaces legacy synchronous OS Win32 file handles with an asynchronous, lock-free, GPU-driven I/O pipeline engineered for NVMe SSDs and DirectX 12.

CPU Bypass

Bypasses traditional system RAM bouncing and CPU thread context switching by routing data straight from NVMe storage to GPU allocations.

GPU Decompression

Parallelizes massive asset decompression (GDeflate, Zstandard) on thousands of GPU compute cores rather than saturating host CPU cores.

Batch Enqueueing

Dispatches tens of thousands of requests per second using lock-free submission queues and standard D3D12 Fence synchronization.

System Requirements & Verification

DirectStorage functions on Windows 10 & 11 with distinct hardware optimization tiers.

Hardware Prerequisites

  • •
    Operating System: Windows 11 provides optimal bypassIO driver optimizations. Windows 10 (v1909 or higher) is supported via fallback compatibility paths.
  • •
    Storage Device: NVMe SSD (PCIe 3.0 / 4.0 / 5.0). Standard SATA SSDs and HDDs function through legacy fallback queues but without high IOPS benefits.
  • •
    Graphics Processor: DirectX 12 compatible GPU with Shader Model 6.0+ support for GPU decompression compute kernels.

Expanding decompression support beyond GDeflate into industry-standard Zstandard (Zstd).

DirectStorage 1.4 Highlights

The 1.4 release introduces Zstandard (Zstd) compression compatibility, allowing game titles to use an open-standard, versatile compressor without sacrificing GPU streaming throughput.

It also integrates D3D12 CreatorIDs to associate DirectStorage resource requests with specific runtime modules for telemetry and profiler isolation, as well as the Game Asset Conditioning Library to streamline build-time packaging.

DirectStorageLoader.cs 
using System;
using System.IO;
using System.Threading;
using Vortice.Direct3D;
using Vortice.Direct3D12;
using Vortice.DirectStorage;
using Vortice.DXGI;

public class DirectStorageLoader : IDisposable
{
    private ID3D12Device? _device;
    private IDStorageFactory? _factory;
    private IDStorageQueue? _queue;

    public void Initialize()
    {
        // 1. Initialize native D3D12 Device
        D3D12.D3D12CreateDevice(null, FeatureLevel.Level_12_0, out _device);
        if (_device == null)
            throw new PlatformNotSupportedException("Failed to create D3D12 Device.");

        // 2. Acquire DirectStorage Singleton Factory
        DStorage.DStorageGetFactory(out _factory);
        if (_factory == null)
            throw new PlatformNotSupportedException("DirectStorage runtime library missing.");

        // 3. Configure Lock-Free Queue for GPU File Streaming
        var queueDesc = new DSTORAGE_QUEUE_DESC
        {
            SourceType = DSTORAGE_REQUEST_SOURCE_TYPE.FILE,
            Capacity = DStorage.DSTORAGE_MAX_QUEUE_CAPACITY,
            Priority = DSTORAGE_PRIORITY.NORMAL,
            Device = _device
        };
        _factory.CreateQueue(queueDesc, out _queue);
    }

    public void StreamFileToGpu(string filePath, ulong byteLength)
    {
        // Open file using DirectStorage specialized handle
        _factory!.OpenFile(filePath, out IDStorageFile? dsFile);

        // Allocate destination GPU resource in Default Heap (VRAM)
        var resourceDesc = new ResourceDescription
        {
            Dimension = ResourceDimension.Buffer,
            Width = byteLength,
            Height = 1,
            DepthOrArraySize = 1,
            MipLevels = 1,
            Format = Format.Unknown,
            SampleDescription = new SampleDescription(1, 0),
            Layout = TextureLayout.RowMajor,
            Flags = ResourceFlags.None
        };

        _device!.CreateCommittedResource(
            new HeapProperties(HeapType.Default),
            HeapFlags.None,
            resourceDesc,
            ResourceStates.Common,
            out ID3D12Resource? gpuBuffer);

        // Create DirectStorage request payload
        var request = new DSTORAGE_REQUEST
        {
            Options = new DSTORAGE_REQUEST_OPTIONS
            {
                SourceType = DSTORAGE_REQUEST_SOURCE_TYPE.FILE,
                DestinationType = DSTORAGE_REQUEST_DESTINATION_TYPE.BUFFER,
                CompressionFormat = DSTORAGE_COMPRESSION_FORMAT.NONE
            },
            Source = new DSTORAGE_SOURCE
            {
                File = new DSTORAGE_SOURCE_FILE { File = dsFile, Offset = 0, Size = (uint)byteLength }
            },
            Destination = new DSTORAGE_DESTINATION
            {
                Buffer = new DSTORAGE_DESTINATION_BUFFER { Resource = gpuBuffer, Offset = 0, Size = (uint)byteLength }
            },
            UncompressedSize = (uint)byteLength
        };

        // Synchronization via D3D12 Fence
        _device.CreateFence(0, FenceFlags.None, out ID3D12Fence? fence);
        ulong targetFenceValue = 1;

        _queue!.EnqueueRequest(request);
        _queue.EnqueueSignal(fence!, targetFenceValue);
        _queue.Submit();

        // Await completed signal safely
        using var waitHandle = new AutoResetEvent(false);
        fence!.SetEventOnCompletion(targetFenceValue, waitHandle.SafeWaitHandle.DangerousGetHandle());
        waitHandle.WaitOne();

        Console.WriteLine("[DirectStorage] Stream completed directly to VRAM.");

        // Cleanup allocations
        dsFile?.Dispose();
        fence?.Dispose();
        gpuBuffer?.Dispose();
    }

    public void Dispose()
    {
        _queue?.Dispose();
        _factory?.Dispose();
        _device?.Dispose();
    }
}

Python ctypes P/Invoke Inspection

Lightweight script to query the native Windows DirectStorage runtime DLL and factory GUID without compiler overhead.

Pure Python
dstorage_probe.py 
import sys
import ctypes
from ctypes import wintypes

# Native COM GUID representation for Windows
class GUID(ctypes.Structure):
    _fields_ = [
        ("Data1", wintypes.DWORD),
        ("Data2", wintypes.WORD),
        ("Data3", wintypes.WORD),
        ("Data4", ctypes.c_ubyte * 8)
    ]

# Interface Identifier: IID_IDStorageFactory
# {5DE95E7B-95B2-486A-86CC-832B52924E57}
IID_IDStorageFactory = GUID(
    0x5DE95E7B, 0x95B2, 0x486A,
    (ctypes.c_ubyte * 8)(0x86, 0xCC, 0x83, 0x2B, 0x52, 0x92, 0x4E, 0x57)
)

def probe_directstorage():
    if sys.platform != "win32":
        print("[ERROR] DirectStorage is supported strictly on Windows 10/11.")
        return False

    try:
        # Attempt to load the official runtime binary
        ds_lib = ctypes.WinDLL("dstoragecore.dll")
    except OSError as err:
        print(f"[FAIL] Could not locate 'dstoragecore.dll': {err}")
        print("[INFO] Ensure Microsoft.DirectStorage redistributable DLL is on PATH or script folder.")
        return False

    # Exported signature: HRESULT DStorageGetFactory(REFIID riid, void** ppv)
    DStorageGetFactory = ds_lib.DStorageGetFactory
    DStorageGetFactory.argtypes = [ctypes.POINTER(GUID), ctypes.POINTER(ctypes.c_void_p)]
    DStorageGetFactory.restype = wintypes.HRESULT

    factory_ptr = ctypes.c_void_p()
    hr = DStorageGetFactory(ctypes.byref(IID_IDStorageFactory), ctypes.byref(factory_ptr))

    S_OK = 0
    if hr == S_OK and factory_ptr.value:
        print("[SUCCESS] DirectStorage Factory acquired successfully!")
        print(f"Pointer address: {hex(factory_ptr.value)}")
        return True
    else:
        print(f"[FAIL] DStorageGetFactory returned HRESULT: {hex(hr & 0xFFFFFFFF)}")
        return False

if __name__ == "__main__":
    probe_directstorage()

High-Performance Python via pybind11 C++ Wrapper

Production-ready C++ extension delivering native GPU asset streaming directly into Python applications.

C++20 / pybind11

DirectStorage utilizes complex C++ COM interfaces and compute shader dispatching. For production Python applications (AI dataset loaders, neural renderers, 3D pipelines), compile a pybind11 C++ dynamic module to expose a straightforward API:

directstorage_pybind.cpp 
#include <pybind11/pybind11.h>
#include <dstorage.h>
#include <d3d12.h>
#include <wrl/client.h>
#include <string>
#include <stdexcept>

namespace py = pybind11;
using Microsoft::WRL::ComPtr;

class DirectStorageStreamer {
public:
    DirectStorageStreamer() {
        // 1. Initialize Direct3D 12 Device
        if (FAILED(D3D12CreateDevice(nullptr, D3D_FEATURE_LEVEL_12_0, IID_PPV_ARGS(&device_)))) {
            throw std::runtime_error("Failed to create D3D12 device");
        }

        // 2. Initialize DirectStorage Factory
        if (FAILED(DStorageGetFactory(IID_PPV_ARGS(&factory_)))) {
            throw std::runtime_error("Failed to initialize DirectStorage Factory");
        }

        // 3. Create File Queue
        DSTORAGE_QUEUE_DESC queueDesc = {};
        queueDesc.SourceType = DSTORAGE_REQUEST_SOURCE_TYPE_FILE;
        queueDesc.Capacity = DSTORAGE_MAX_QUEUE_CAPACITY;
        queueDesc.Priority = DSTORAGE_PRIORITY_NORMAL;
        queueDesc.Device = device_.Get();

        if (FAILED(factory_->CreateQueue(&queueDesc, IID_PPV_ARGS(&queue_)))) {
            throw std::runtime_error("Failed to create DirectStorage Queue");
        }

        // 4. Initialize Fence for synchronization
        device_->CreateFence(0, D3D12_FENCE_FLAG_NONE, IID_PPV_ARGS(&fence_));
        fenceEvent_ = CreateEvent(nullptr, FALSE, FALSE, nullptr);
    }

    virtual ~DirectStorageStreamer() {
        if (fenceEvent_) CloseHandle(fenceEvent_);
    }

    bool stream_file(const std::string& path, uint64_t fileSize) {
        ComPtr<IDStorageFile> file;
        std::wstring wpath(path.begin(), path.end());

        if (FAILED(factory_->OpenFile(wpath.c_str(), IID_PPV_ARGS(&file)))) {
            return false;
        }

        // Prepare committed GPU buffer in Default Heap
        D3D12_HEAP_PROPERTIES heapProps = { D3D12_HEAP_TYPE_DEFAULT };
        D3D12_RESOURCE_DESC resDesc = {};
        resDesc.Dimension = D3D12_RESOURCE_DIMENSION_BUFFER;
        resDesc.Width = fileSize;
        resDesc.Height = 1;
        resDesc.DepthOrArraySize = 1;
        resDesc.MipLevels = 1;
        resDesc.SampleDesc.Count = 1;
        resDesc.Layout = D3D12_TEXTURE_LAYOUT_ROW_MAJOR;

        ComPtr<ID3D12Resource> gpuBuffer;
        device_->CreateCommittedResource(
            &heapProps,
            D3D12_HEAP_FLAG_NONE,
            &resDesc,
            D3D12_RESOURCE_STATE_COMMON,
            nullptr,
            IID_PPV_ARGS(&gpuBuffer));

        // Enqueue Read Request
        DSTORAGE_REQUEST request = {};
        request.Options.SourceType = DSTORAGE_REQUEST_SOURCE_TYPE_FILE;
        request.Options.DestinationType = DSTORAGE_REQUEST_DESTINATION_TYPE_BUFFER;
        request.Options.CompressionFormat = DSTORAGE_COMPRESSION_FORMAT_NONE;
        request.Source.File.File = file.Get();
        request.Source.File.Offset = 0;
        request.Source.File.Size = static_cast<uint32_t>(fileSize);
        request.Destination.Buffer.Resource = gpuBuffer.Get();
        request.Destination.Buffer.Size = static_cast<uint32_t>(fileSize);
        request.UncompressedSize = static_cast<uint32_t>(fileSize);

        queue_->EnqueueRequest(&request);
        
        // Enqueue fence signal & submit
        uint64_t waitVal = ++fenceVal_;
        queue_->EnqueueSignal(fence_.Get(), waitVal);
        queue_->Submit();

        // Block until GPU DMA finishes
        fence_->SetEventOnCompletion(waitVal, fenceEvent_);
        WaitForSingleObject(fenceEvent_, INFINITE);
        return true;
    }

private:
    ComPtr<ID3D12Device> device_;
    ComPtr<IDStorageFactory> factory_;
    ComPtr<IDStorageQueue> queue_;
    ComPtr<ID3D12Fence> fence_;
    HANDLE fenceEvent_ = nullptr;
    uint64_t fenceVal_ = 0;
};

// Expose class directly to Python using pybind11
PYBIND11_MODULE(py_directstorage, m) {
    m.doc() = "High-performance DirectStorage Python Bridge";
    py::class_<DirectStorageStreamer>(m, "DirectStorageStreamer")
        .def(py::init<>())
        .def("stream_file", &DirectStorageStreamer::stream_file, 
             py::arg("path"), py::arg("file_size"));
}

Performance & Benchmarking Guidelines

Best practices for saturating PCIe bandwidth and preventing pipeline stalls.

1. Batch Submissions

Do not submit individual 4KB reads. Enqueue hundreds of requests with EnqueueRequest() before calling Submit() to maximize NVMe queue depth.

2. Choose GDeflate/Zstd

Use tile-compressed GDeflate or Zstd packages. GPU compute clusters decompress gigabytes in milliseconds, turning the NVMe transfer rate into the sole factor.

3. Enable BypassIO

Ensure the target drive uses NTFS on Windows 11 with driver stack filters compatible with BypassIO to avoid transitioning to legacy OS filter managers.
